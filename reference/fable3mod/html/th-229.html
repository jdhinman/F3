use simple code:
if Inventory.GetNumberOfItemsOfType(GetLocalHero(), 'ObjectInventoryPotionDogStatsLevel5') > 0 then

DogStats.SetTrainingLevel(GetDog(GetLocalHero()), EDogTrainingType.DOG_TRAINING_TYPE_COMBAT, 0)
