try
local allObjects = {
	'ObjectTattooSuitSE1',
	'ObjectTattooSuitSE2',
	'ObjectTattooSuitSE3'
	--etc
}

for key,value in pairs(allObjects) do
	if Inventory.GetNumberOfItemsOfType(GetLocalHero(), value) == 0 then
		Inventory.AddItemOfType(GetLocalHero(), value)
		GUI.DisplayReceivedItem(value)
	end
end
