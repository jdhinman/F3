#pragma byteorder(big_endian)
#pragma maxarray(65536) 	// max that hw 5.1 will allow
#pragma maxstring(512)

typedef struct GENV		// environment map
{
	DWORD	Unknowns1[5];
	DWORD	CellsX;
	DWORD	CellsY;
	DWORD	Unknown2;
	
	DWORD	Map[CellsX*CellsY];	// Environment ID for that cell
} GENV;
